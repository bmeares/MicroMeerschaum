cmake_minimum_required(VERSION 3.16)
project(micromeerschaum_core C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

option(MM_SANITIZE "Build host tests with ASan and UBSan" ON)

if(MM_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-sanitize-recover=all)
endif()

add_library(micromeerschaum src/crc32.c src/wire.c src/record.c src/segment.c src/batch.c
            src/protocol.c src/document.c src/pipe.c src/gatt.c src/actions.c src/action_shell.c src/pipe_actions.c)
target_include_directories(micromeerschaum PUBLIC include)
target_compile_options(micromeerschaum PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
if(MM_SANITIZE)
    target_link_options(micromeerschaum PUBLIC -fsanitize=address,undefined)
endif()

enable_testing()
foreach(test vectors roundtrip protocol document gatt actions)
    add_executable(test_${test} tests/test_${test}.c)
    target_link_libraries(test_${test} PRIVATE micromeerschaum)
    target_compile_options(test_${test} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME ${test} COMMAND test_${test})
endforeach()
target_compile_definitions(test_vectors PRIVATE MM_VECTORS_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../vectors")
target_compile_definitions(test_protocol PRIVATE MM_VECTORS_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../vectors")
target_compile_definitions(test_document PRIVATE MM_VECTORS_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../vectors")

if(UNIX)
    enable_language(CXX)
    add_library(micromeerschaum_posix storage/journal.c storage/volume.c posix/peer.c shared/sender.c)
    target_link_libraries(micromeerschaum_posix PUBLIC micromeerschaum)
    target_compile_options(micromeerschaum_posix PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)

    foreach(test journal delivery sender volume custody pipe_actions)
        add_executable(test_${test} tests/test_${test}.c)
        target_link_libraries(test_${test} PRIVATE micromeerschaum_posix)
        target_compile_options(test_${test} PRIVATE -Wall -Wextra -Wpedantic -Werror)
        add_test(NAME ${test} COMMAND test_${test})
    endforeach()

    add_executable(test_pipe tests/test_pipe.cpp)
    target_compile_features(test_pipe PRIVATE cxx_std_17)
    target_link_libraries(test_pipe PRIVATE micromeerschaum_posix)
    target_compile_options(test_pipe PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-exceptions)
    add_test(NAME pipe COMMAND test_pipe)

    add_executable(mm_temperature ../examples/host/temperature.cpp)
    target_compile_features(mm_temperature PRIVATE cxx_std_11)
    target_link_libraries(mm_temperature PRIVATE micromeerschaum_posix)
    target_compile_options(mm_temperature PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-exceptions)

    add_executable(mm_gatt_test_node tests/gatt_node.c)
    target_link_libraries(mm_gatt_test_node PRIVATE micromeerschaum_posix)
    target_compile_options(mm_gatt_test_node PRIVATE -Wall -Wextra -Wpedantic -Werror)

    add_executable(mm_action_shell ../examples/host/actions.c)
    target_link_libraries(mm_action_shell PRIVATE micromeerschaum)
    target_compile_options(mm_action_shell PRIVATE -Wall -Wextra -Wpedantic -Werror)

    add_executable(mm_sync_example ../examples/host/sync.c)
    target_link_libraries(mm_sync_example PRIVATE micromeerschaum_posix)
    target_compile_options(mm_sync_example PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

option(MM_WITH_SQLITE "Build the optional local SQLite instance" OFF)
if(MM_WITH_SQLITE)
    include(sqlite/engine.cmake)
    add_library(micromeerschaum_sqlite sqlite/instance.c sqlite/write.c sqlite/rows.c sqlite/outbox.c)
    target_link_libraries(micromeerschaum_sqlite PUBLIC micromeerschaum PRIVATE mm_sqlite_engine)
    target_compile_options(micromeerschaum_sqlite PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
    add_executable(test_sqlite tests/test_sqlite.c)
    target_link_libraries(test_sqlite PRIVATE micromeerschaum_sqlite)
    add_test(NAME sqlite COMMAND test_sqlite)
endif()
if(MM_WITH_SQLITE AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(test_sqlite_recovery tests/test_sqlite_recovery.c)
    target_link_libraries(test_sqlite_recovery PRIVATE micromeerschaum_sqlite)
    target_link_options(test_sqlite_recovery PRIVATE -Wl,--wrap=fsync -Wl,--wrap=pwrite64)
    add_test(NAME sqlite_recovery COMMAND test_sqlite_recovery)
endif()

# Test the same JSON parser used by the ESP-IDF json component on the host.
set(MM_CJSON_DIR "" CACHE PATH "Optional directory containing the SDK's cJSON.c and cJSON.h")
if(MM_CJSON_DIR)
    add_library(mm_device_session ../ports/esp-idf/device_session.c "${MM_CJSON_DIR}/cJSON.c")
    target_include_directories(mm_device_session PUBLIC "${MM_CJSON_DIR}")
    target_link_libraries(mm_device_session PUBLIC micromeerschaum)
    if(UNIX)
        target_link_libraries(mm_device_session PRIVATE m)
    endif()
    target_compile_options(mm_device_session PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_executable(test_device_session tests/test_device_session.c)
    target_link_libraries(test_device_session PRIVATE mm_device_session)
    add_test(NAME device_session COMMAND test_device_session)
endif()
